record Complex : Set where
  constructor _+i_
  field
    re : Float
    im : Float
open Complex

_⊕_ : Complex → Complex → Complex
a ⊕ b = (re a + re b) +i (im a + im b)

_⊖_ : Complex → Complex → Complex
a ⊖ b = (re a - re b) +i (im a - im b)

_⊗_ : Complex → Complex → Complex
a ⊗ b = (re a * re b - im a * im b) +i (re a * im b + im a * re b)

cis : Float → Complex
cis t = cos t +i sin t

pi : Float
pi = 3.141592653589793
