structure Complex where
  re : Float
  im : Float

instance : Add Complex := ⟨fun a b => ⟨a.re + b.re, a.im + b.im⟩⟩
instance : Sub Complex := ⟨fun a b => ⟨a.re - b.re, a.im - b.im⟩⟩
instance : Mul Complex := ⟨fun a b => ⟨a.re * b.re - a.im * b.im, a.re * b.im + a.im * b.re⟩⟩

def Complex.cis (t : Float) : Complex := ⟨Float.cos t, Float.sin t⟩

def pi : Float := 3.141592653589793

partial def fft : List Complex → List Complex
